# Emulator definitions. Run by t1-definitions.ps1, which makes the test
# folder T (fake emulators, the user's own definitions in <data>\definitions, synthetic game headers).
#  - Zeta is known only from a user definition file: found by itself, plays its game.
#  - A user file changes a shipped definition (Dolphin's name), the rest of the shipped one stays (Managed).
#  - Old Emu has a real version (a copy of where.exe) outside its tested range: one level lower.
#  - A broken user file is skipped, one with null lists or groups loads empty (or is skipped: no programs).
#  - Old Saves (version outside its range, save folders): Generic, but saves stay per profile and no seed file.
#  - The game id readers on synthetic headers.
#  - Imported definitions: Stella (from Playnite) is found, a core from libretro-core-info has its firmware list,
#    and hand-made definitions win over imported patterns for the same program.
wait 900
type Tester
key accept
wait 900
key accept
wait 2200
emulog
expect emus 7
expect defname zeta|Zeta
expect defname dolphin|Dolphin (user)
expect defname shadps4|shadPS4
expect defname suyu|suyu
expect emulevel Zeta|Known
expect emulevel Dolphin (user)|Managed
expect emulevel Old Emu|Known
expect emulevel Old Saves|Generic
expect defname nullsys|Null Systems
expect emulevel Null Systems|Known
expect emulevel Stella|Known
expect deffiles stella|0|0|source
expect deffiles fceumm-core|3|2|source
expect deffiles dolphin|0|0|hand
expect defforexe shadPS4.exe|shadps4
expect defforexe Ryujinx.exe|ryujinx-1
expect defforexe fceumm_libretro.dll|fceumm-core
expect defforexe setup.exe|none
expect filehas T\emus\Dolphin\User\Config\Dolphin.ini|ConfirmStop = False
expect gameid nds-header|T\ids\test.nds|ATSE|1|IONEA TEST
expect gameid gba-header|T\ids\test.gba|BTSE|2|IONEA GBA
expect gameid gb-header|T\ids\test.gb|IONEAGB|3|IONEAGB
expect gameid gc-disc|T\ids\test.iso|GTSE01|4|Ionea Cube
expect gameid gc-disc|T\ids\test.wbfs|RTSE01|0|Ionea Wii
expect gameid 3ds-ncsd|T\ids\test.3ds|0004000000123400|5|CTR-P-ITSE
expect gameid ps4-sfo|T\ids\Ionea Test [CUSA12345]\eboot.bin|CUSA12345|01.05|Ionea PS4
expect gameid ps4-sfo|T\ids\Ionea Two\eboot.bin|CUSA54321|01.00|Ionea Two
expect gameid psp-pbp|T\ids\test.pbp|ULUS12345|1.01|Ionea PSP
expect gameid wiiu-meta|T\ids\wiiu\code\ionea.rpx|0005000010ABCD00|32|Ionea U
expect gameid nds-header|T\ids\not-a-game.nds|none
# a game for the emulator that only has a user definition
addrom T\roms\Zeta Test.nds
wait 600
expect screen AddRomCard
key accept
wait 900
expect name Zeta Test
expect emu Zeta
select Zeta Test
key accept
wait 300
waitlaunch 25
expect launch shown
expect played yes
gamesystem Zeta Test|nds
# a game for the NES emulator (the cores run checks it keeps its emulator)
romgame NES Test|T\roms\NES Test.nes|Nes Emu
gamesystem NES Test|nes
# a dropped version keeps saves per profile
romgame Old Saves Game|T\roms\Old Saves Game.iso|Old Saves
prepare Old Saves Game|Old Saves
expect link T\emus\OldSaves\Saves
expect filehas T\emus\OldSaves\Saves\old.sav|before
newprofile
type Sam
key accept
wait 700
expect profile Sam
prepare Old Saves Game|Old Saves
expect gone T\emus\OldSaves\Saves\old.sav
writefile T\emus\OldSaves\Saves\sam.sav|sam save
switchprofile Tester
prepare Old Saves Game|Old Saves
expect filehas T\emus\OldSaves\Saves\old.sav|before
expect gone T\emus\OldSaves\Saves\sam.sav
expect gone T\emus\OldSaves\seeded.ini
quit
