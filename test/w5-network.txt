# Network control: a new profile starts in Offline mode with every other network row greyed,
# nothing goes out while it is on (even with a source switched on), a host outside the source is refused,
# the Discord status follows Offline mode, and the activity log shows each attempt and clears.
# Setup: test\w5-network.ps1 (12 dummies, --windowed --mute, off screen). No semicolons in comments.
wait 900
type Tester
key accept
wait 900
key accept
wait 2200
expect offline on
expect discordwish off
settings network
expect settingspane Network
key right
expect cardfocus Offline mode
expect rowvalue Offline mode|On
expect hosts emulators|github.com, githubusercontent.com
expect hosts covers|thumbnails.libretro.com
shot w5-network-offline
# a greyed switch does nothing while offline
key down
expect cardfocus Covers
key right
expect rowvalue Covers|Off
wait 300
key back 2
wait 600
# a source switched on still makes no request while offline
online covers on
download covers|https://thumbnails.libretro.com/x.png|T\dl\x.png
expect onlinelast covers|blocked
expect onlinerequests 0
# offline off: the Discord status may show, and a host outside the source is refused before any request
offline off
expect offline off
expect discordwish home
download covers|https://example.com/x.png|T\dl\y.png
expect onlinelast covers|blocked
expect onlinerequests 0
expect onlinelog 2
settings network
key right
expect rowvalue Offline mode|Off
expect rowvalue Covers|On
shot w5-network-online
key back 2
wait 600
# the activity log, then cleared
onlinelog
expect screen OnlineLogCard
shot w5-activity
cardrow Clear log
key accept
wait 800
expect onlinelog 0
expect screen OnlineLogCard
shot w5-activity-cleared
key back
wait 500
offline on
expect discordwish off
quit
