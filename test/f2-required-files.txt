# Required files. Run by f2-required-files.ps1, which makes the test folder T:
# fake emulators with the user's own definitions, made-up key / firmware / BIOS files (right ones and wrong ones).
#  - First start (--files-prompt): the Required files overview lists the emulators that lack a needed file.
#  - Missing shows Missing, a wrong file (bad structure, bad hash) is refused and nothing is copied.
#  - A right file (from a zip, a folder, a single file) is copied in, shows Ready, and is kept once in the store,
#    Swit B (same family) gets the same files, Own Emu's own file counts as Ready and is never replaced.
#  - A folder entry with hashes, an emulator that installs its firmware itself, a core's system folder.
wait 900
type Tester
key accept
wait 900
key accept
wait 2200
expect screen RequiredOverviewCard
expect rowvalue Swit A|3 missing
expect rowvalue Swit B|3 missing
expect rowvalue Psx Emu|1 missing
expect rowvalue Pup Emu|1 missing
expect norow Own Emu
expect norow melonDS DS core
shot f2-overview
key back
wait 600
expect screen HomeScreen

# Ready / Missing / Optional, and the in-place file
expect reqstate Swit A|prod.keys|Missing
expect reqstate Swit A|title.keys|Optional
expect reqstate Swit A|Firmware|Missing
expect reqstate Own Emu|prod.keys|Ready
expect reqmissing Own Emu|0
expect reqstate melonDS DS core|bios7.bin|Optional
reqfiles Swit A
expect screen RequiredFilesCard
expect rowvalue prod.keys|Missing
expect rowvalue title.keys|Optional
shot f2-missing
key back
wait 400

# wrong files are refused
reqpick Swit A|T\bad\prod.keys|prod.keys
wait 900
expect screen MessageCard
key back
wait 300
expect reqstate Swit A|prod.keys|Missing
reqpick Swit A|T\bad\boot.bin
wait 900
expect screen MessageCard
key back
wait 300
expect reqstate Swit A|boot.bin|Missing
expect gone T\emus\SwitA\sys\boot.bin
reqpick melonDS DS core|T\bad\boot.bin|bios7.bin
wait 900
expect screen MessageCard
key back
wait 300
expect gone T\emus\Cores\system\bios7.bin

# a zip with the right keys (in a subfolder) and the right boot.bin
reqpick Swit A|T\good.zip
wait 1500
expect screen RequiredFilesCard
expect reqstate Swit A|prod.keys|Ready
expect reqstate Swit A|boot.bin|Ready
expect toast Ready | Swit A: prod.keys, boot.bin
expect rowvalue prod.keys|Ready
shot f2-ready
expect reqstate Swit B|prod.keys|Ready
expect reqstate Swit B|boot.bin|Ready
expect reqmissing Swit B|1
expect reqstate Own Emu|prod.keys|Ready

# a folder: the firmware
reqpick Swit A|T\fw|Firmware
wait 1500
expect reqstate Swit A|Firmware|Ready
expect reqstate Swit B|Firmware|Ready
expect reqmissing Swit A|0
expect reqmissing Swit B|0
key back
wait 300

# a row pick replaces what is there (the old copy is kept)
reqpick Swit A|T\good2\prod.keys|prod.keys
wait 900
expect reqstate Swit A|prod.keys|Ready
key back
wait 300

# a folder entry with a list of hashes: the right file is found deep in the picked folder
reqpick Psx Emu|T\psxdir
wait 900
expect reqstate Psx Emu|PS1 BIOS|Ready
key back
wait 300

# the emulator installs it itself (a copy of cmd.exe copies the fake .PUP)
reqpick Pup Emu|T\bad\boot.bin|Firmware (.PUP)
wait 900
expect screen MessageCard
key back
wait 300
reqpick Pup Emu|T\PS3UPDAT.PUP|Firmware (.PUP)
wait 3000
expect reqstate Pup Emu|Firmware (.PUP)|Ready
key back
wait 300

# the emulator's own screen
emucard Swit A
expect rowvalue Required files|Ready
key back
wait 300
emucard Psx Emu
expect rowvalue Required files|Ready
key back
wait 300
reqoverview
expect cardrows Done
key back
wait 300
quit
