# The launcher's own file and folder browser (Picker.cs), as a controller user gets it. Run by b2-browser.ps1.
wait 900
type Tester
key accept
wait 900
key accept
wait 2200
expect screen HomeScreen
# a file: folders first, then only the files that fit
pick file T\browse\Games .iso,.chd
expect screen FileBrowserCard
expect browserdir Games
expect browserrows saves | Alpha.iso | Beta.chd
shot b2-files
browse Beta.chd
expect browserpicked Beta.chd
expect browserdir closed
expect screen HomeScreen
# a folder: B goes up a folder, focused on the one it came from
pick folder T\browse\Games
expect browserrows Choose this folder | saves
key back
expect browserdir browse
expect browserrows Choose this folder | Empty | Games
expect cardfocus Games
browse Empty
expect browserrows Choose this folder
shot b2-empty
browse Choose this folder
expect browserpicked Empty
# with no start folder: This PC, and B there closes without choosing anything
pick file none
expect browserdir This PC
key back
expect browserdir closed
expect browserpicked none
expect screen HomeScreen
quit
