# The scan, header identification, covers. Run by s7-scan.ps1 with
# --scan-home <T>\home (a fake home) and --online-test-host localhost:47812.
wait 900
type Tester
key accept
wait 900
key accept
wait 2200
expect screen HomeScreen
expect emus 0
expect games 0
expect esde none
expect inbox none

# Header identification: the file says its system, whatever its name.
expect systemof T\home\Games\Mixed\Mystery.bin|gba
expect systemof T\home\Games\GBA\Pocket Quest (USA).gba|gba
expect systemof T\home\ES-DE\ROMs\nds\Star Racer (Europe).nds|nds
expect systemof T\home\Games\Switch\Island Trip.nro|switch
expect systemof T\home\Games\Mixed\Unknown.iso|none

# The scan: everything ticked.
scan
waitscan 30
scanlog
expect screen ScanCard
shot s7-results
expect scanhas PPSSPP=Add
expect scanhas Eden=Add
expect scanhas DuckStation=Add
expect scanhas melonDS=Add
expect scanhas ES-DE=Use
expect scanhas 5 games=Add
expect scannot Dolphin
expect scannot ROMs\gc
expect scanhas Look harder
expect cardfocus Add
scantoggle DuckStation
expect scanhas DuckStation=Skip
scanconfirm
wait 800
expect screen HomeScreen
expect emus 4
expect games 4
expect gamesystem Mystery|gba
expect gamesystem Pocket Quest|gba
expect gamesystem Star Racer|nds
expect gamesystem Island Trip|switch
expect gamefolder T\home\Games\PS1
expect gamefolder T\home\Games\Mixed
expect esde T\home\ES-DE
shot s7-home

# Covers: nothing while switched off, then LibRetro by game code or CRC in the .dat, never by file name.
# Offline mode is on for a new profile: off here, so the source switches decide.
offline off
expect requests 0
coverfetch
waitcovers 5
expect requests 0
expect coverfrom Pocket Quest|none|no cover
online covers on
online names on
coverfetch
waitcovers 30
expect coverfrom Pocket Quest|libretro|cover
expect coverfrom Mystery|libretro|cover
expect coverfrom Star Racer|none|no cover

# Looking again: only what is still new.
scan
waitscan 30
scanlog
expect scanhas DuckStation=Add
expect scannot PPSSPP
expect scannot games
key back
wait 500
expect screen HomeScreen

# Change cover: SteamGridDB with the user's key, then a pick, and the name tile stays a name tile.
online steamgriddb on
sgdbkey s7-test-key
covercard Star Racer
waitcovercard 20
expect screen CoverPickCard
expect coversources steamgriddb | name | file
shot s7-cover-pick
coverpicksrc steamgriddb
expect coverfrom Star Racer|own|cover
covercard Pocket Quest
waitcovercard 20
coverpicksrc name
expect coverfrom Pocket Quest|name|no cover
coverfetch
waitcovers 10
expect coverfrom Pocket Quest|name|no cover
quit
