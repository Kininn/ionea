# k3: the launcher's own keys. Defaults (WASD and the arrows, Enter / Space, Esc, Alt for the
# Home menu), moving with WASD, a held Select key, the hints, the Launcher keys cards, the Home menu key
# taken from game buttons (and the notice), Esc held as a game button, and the Home menu key's press logic.
# No real input: "keydown" / "rawkey" call the launcher, "hooksim" runs the hook's logic only.
# Run: --data <tmp> --no-import --dummies 6 --windowed --mute --script test\k3-launcher-keys.txt
wait 900
type Keys Test
key accept
wait 900
mouse down 900 500
wait 1100
expect screen HomeScreen
# ---- defaults ----
expect launcherkey up|W  ·  ↑
expect launcherkey down|S  ·  ↓
expect launcherkey left|A  ·  ←
expect launcherkey right|D  ·  →
expect launcherkey accept|Enter  ·  Space
expect launcherkey back|Esc  ·  Backspace
expect launcherkey home|Alt
expect keymap W|Up
expect keymap S|Down
expect keymap A|Left
expect keymap D|Right
expect keymap Up|Up
expect keymap Return|Accept
expect keymap Space|Accept
expect keymap Escape|Back
expect keymap Back|Back
expect keymap LeftAlt|none
# ---- WASD moves on home, a held Space opens the quick menu ----
expect sel 0
keydown D
keyup D
expect sel 1
keydown A
keyup A
expect sel 0
keydown Right
expect sel 1
keydown Left
expect sel 0
expect hinttext X Options  Enter Start
keydown Space
wait 1000
expect screen HoldMenu
keyup Space
keydown Escape
wait 400
expect screen HomeScreen
# hints follow the keys

expect hinttext F2 Options  Enter Start
launcherkey option default
expect hinttext X Options  Enter Start
# ---- Settings > Input > Launcher keys ----
launcherkeys
expect screen LauncherKeysCard
shot k3-launcher-keys
key back
wait 300
# one action's card: add, refuse, move from another action, cancel, remove
launcherkeycard up
expect screen LauncherKeyCard
cardrow Add a key
key accept
expect lkcapturing yes
rawkey LeftAlt
expect lkmsg Alt can't be a launcher key  ·  pick another key, or Esc
expect lkcapturing yes

expect lkcapturing no
expect lkmsg Q set
expect launcherkey up|W  ·  ↑  ·  Q
expect keymap Q|Up
cardrow Add a key
key accept
rawkey S
expect lkmsg S set  ·  taken from Down
expect launcherkey down|↓
expect keymap S|Up
shot k3-launcher-key-up
# four keys is the most
expect launcherkey up|W  ·  ↑  ·  Q  ·  S
cardrow Q
key accept
expect lkmsg Q removed
expect launcherkey up|W  ·  ↑  ·  S
cardrow Add a key
key accept
rawkey Escape
wait 600
expect lkcapturing no
expect lkmsg Cancelled
expect launcherkey up|W  ·  ↑  ·  S
key back
wait 300
# Back keeps a key, and its last one can't be taken by another action
launcherkey back Escape
launcherkeycard back
cardrow Esc
key accept
expect launcherkey back|Esc
expect lkmsg Back needs a key
key back
wait 300
launcherkeycard up
cardrow Add a key
key accept
keyheld Escape
rawkey Escape
wait 600
keyheld none
expect lkmsg Esc is the only Back key  ·  pick another key, or Esc
expect launcherkey back|Esc

key back

expect launcherkey back|Esc  ·  Backspace

hooksim up Escape

hooksim up RightCtrl

wait 300
expect homefires 3
homeclaim off
hooksim down LeftCtrl
hooksim up LeftCtrl
expect hooksim pass
