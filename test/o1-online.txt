# Network. Run by o1-online.ps1, which serves http://localhost:47811
# and writes down every request it gets. First start screen (all off), no request while a switch is off,
# only the item's own site, redirects checked, a program asks every time and shows the address, the log,
# Settings > Network. Ends by forgetting one answer, and o1-online-restart.txt checks it is asked again.
wait 900
type Tester
key accept
wait 1100
expect screen StartScreen
key accept
wait 2200
expect screen OnlineCard
expect onlineasked no
expect cardrows Turn everything on | Offline mode | Covers | Game names database | Emulator installs | Cores | Definition updates | Covers from SteamGridDB | Launcher updates | Done
expect rowvalue Offline mode|On
expect rowvalue Covers|Off
expect rowvalue Game names database|Off
expect rowvalue Emulator installs|Off
expect rowvalue Cores|Off
expect rowvalue Definition updates|Off
expect rowvalue Covers from SteamGridDB|Off
expect cardfocus Turn everything on
shot o1-first-start
key down 9
expect cardfocus Done
key accept
wait 800
# --setup: the controls question comes right after
expect screen SetupPromptCard
key back
wait 500
# Offline mode is on for a new profile: off here, so the switches decide
offline off
expect screen HomeScreen
expect onlineasked yes
expect online covers|off
expect online names|off
expect online emulators|off
expect online cores|off
expect online definitions|off
expect online steamgriddb|off

# Everything off: each kind is refused and nothing is asked for.
download covers|http://localhost:47811/cover.png|T\dl\cover.png
waitdownload 5
expect download blocked
download names|http://localhost:47811/names.dat|T\dl\names.dat
waitdownload 5
expect download blocked
download emulators|http://localhost:47811/emu.zip|T\dl\emu.zip|Test Emu
waitdownload 5
expect download blocked
expect screen HomeScreen
download cores|http://localhost:47811/core.dll|T\dl\core.dll|Test core
waitdownload 5
expect download blocked
expect screen HomeScreen
download definitions|http://localhost:47811/defs.json|T\dl\defs.json
waitdownload 5
expect download blocked
expect onlinerequests 0
expect onlinelog 5
expect onlinelast definitions|blocked
expect gone T\dl\cover.png

# Covers switched on in Settings > Network.
settings network
expect settingspane Network
expect cardfocus Network
expect rowvalue Covers|Off
expect rowvalue Network activity|5
shot o1-settings-off
key right
expect cardfocus Offline mode
key down
expect cardfocus Covers
key right
expect online covers|on
expect rowvalue Covers|On
shot o1-settings-covers-on
key back 2
wait 400
expect screen HomeScreen

download covers|http://localhost:47811/cover.png|T\dl\cover.png
waitdownload 10
expect download done
expect onlinerequests 1
expect filehas T\dl\cover.png|IONEA-COVER
# on, but not its site, or not https: refused before any request
download covers|https://example.com/x.png|T\dl\x.png
waitdownload 5
expect download blocked
download covers|http://thumbnails.libretro.com/x.png|T\dl\x.png
waitdownload 5
expect download blocked
expect onlinerequests 1
# sent on to another address: checked again (same site: fine, elsewhere: stopped)
download covers|http://localhost:47811/moved|T\dl\moved.png
waitdownload 10
expect download done
expect onlinerequests 3
expect filehas T\dl\moved.png|IONEA-COVER
download covers|http://localhost:47811/away|T\dl\away.png
waitdownload 10
expect download failed
expect onlinerequests 4
expect gone T\dl\away.png

# A program asks every time and shows the address, even with its switch on. Cancel is first.
online cores on
download cores|http://localhost:47811/core.dll|T\dl\core.dll|Test core
wait 300
expect screen DownloadCard
expect download running
expect downloadaddress http://localhost:47811/core.dll
expect cardfocus Cancel
shot o1-program-question
key accept
waitdownload 5
expect download cancelled
expect onlinerequests 4
download cores|http://localhost:47811/core.dll|T\dl\core.dll|Test core
wait 300
expect screen DownloadCard
key back
waitdownload 5
expect download cancelled
expect onlinerequests 4
expect gone T\dl\core.dll
download cores|http://localhost:47811/core.dll|T\dl\core.dll|Test core
wait 300
key down
expect cardfocus Download
key accept
waitdownload 10
expect download done
expect onlinerequests 5
expect filehas T\dl\core.dll|IONEA-CORE
# emulators still off: refused, no question
download emulators|http://localhost:47811/emu.zip|T\dl\emu.zip|Test Emu
waitdownload 5
expect download blocked
expect screen HomeScreen
expect onlinerequests 5

# The log: what, from where, when.
expect onlinelog 14
expect onlinelast emulators|blocked
onlinelog
expect screen OnlineLogCard
shot o1-log
key back
wait 400

# The same screen again: answers kept.
onlinecard
expect screen OnlineCard
expect rowvalue Covers|On
expect rowvalue Cores|On
expect rowvalue Emulator installs|Off
key back
wait 400
expect screen HomeScreen
# As if "Definition updates" were new in the next version.
onlineforget definitions
expect onlineasked no
quit
