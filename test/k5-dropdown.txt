# k5: settings with three or more choices open a list of them. Layout, a long list (the test
# card's None and ten games: the list shows eight at a time), keys and the mouse in the list, Left / Right
# still stepping. No real input.
# Run: --data <tmp> --no-import --dummies 10 --windowed --mute --script test\k5-dropdown.txt (run11.ps1)
wait 2500
type List Test
key accept
wait 900
mouse down 900 500
wait 1100
expect screen HomeScreen
expect zoom Large
# ---- Settings > Look and layout > Layout ----
settings
expect settingspane Look and layout
key right
expect settingsrail no
cardrow Layout
expect cardfocus Layout
expect list closed
key accept
expect list Large|Medium|Small
expect listfocus Large
key down
expect listfocus Medium
key down
key down
expect listfocus Small
shot k5-layout-list
key back
expect list closed
expect zoom Large
expect screen SettingsCard
key accept
key down
key accept
wait 300
expect list closed
expect zoom Medium
expect rowvalue Layout|Medium
# Left / Right still step without the list
key right
expect list closed
expect zoom Small
key left
expect zoom Medium
# the mouse: a click on the row opens, pointing moves, a click picks, a click outside closes
mouserow down Layout
expect list Large|Medium|Small
mouserow up Small
expect list Large|Medium|Small
mouserow move Small
expect listfocus Small
mouserow down Small
wait 300
expect list closed
expect zoom Small
mouserow down Layout
expect listfocus Small
mouse wheelup
expect listfocus Medium
mouse down 5 5
expect list closed
expect screen SettingsCard
expect zoom Small
mouserow down Layout
listpick Large
expect zoom Large
expect list closed
# Back with the list open closes only the list
mouserow down Layout
key back
expect list closed
expect screen SettingsCard
key back 2
wait 400
expect screen HomeScreen
# ---- a long list (ListTestCard, the profile's Favourite game once) ----
listcard
expect screen ListTestCard
cardrow Game
key accept
expect listfocus None
expect listshown None|Crystal Quest|Frozen Garden|Golden Racers|Hidden Circuit|Iron Signal|Lunar Rally|Neon Drift  (1 of 11)
key down 9
expect listshown Frozen Garden|Golden Racers|Hidden Circuit|Iron Signal|Lunar Rally|Neon Drift|Silent Kingdom|Solar Frontier  (10 of 11)
key pageleft
expect listfocus Crystal Quest
key pageright
expect listfocus Solar Frontier
shot k5-long-list
key accept
wait 300
expect list closed
expect rowvalue Game|Solar Frontier
key accept
expect listfocus Solar Frontier
mouse wheeldown
mouse wheeldown
expect listshown Golden Racers|Hidden Circuit|Iron Signal|Lunar Rally|Neon Drift|Silent Kingdom|Solar Frontier|Wild Runner  (11 of 11)
key up 10
expect listfocus None
key back
expect rowvalue Game|Solar Frontier
key back
key back
wait 400
quit
