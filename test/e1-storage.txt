# Moving games between places: across drives and back, a cue sheet and its tracks, the same drive,
# a name clash, a cancelled move.
# Set up first, and again before every run: python -I test\e1-storage-reset.py [--other <folder on another drive>]
# It prints the environment to set and the command line that runs this script.
wait 2500
storage
shot storage
key back
move Game A|Other drive
waitmove 60
expect move done
expect rom Game A|T2\gba\a.gba
expect exists T2\gba\a.sav
expect gone T\pc\gba\a.gba
expect gone T\pc\gba\a.sav
expect exists T\pc\gba\ab.gba
move Game A|PC
waitmove 60
expect move done
expect rom Game A|T\pc\gba\a.gba
expect exists T\pc\gba\a.sav
expect gone T2\gba\a.gba
move Game B|Other drive
waitmove 60
expect move done
expect rom Game B|T2\psx\b.cue
expect exists T2\psx\b (Track 1).bin
expect exists T2\psx\b (Track 2).bin
expect gone T\pc\psx\b (Track 2).bin
move Game C|PC2
waitmove 30
expect move done
expect rom Game C|T\pc2\nds\c.nds
expect gone T\pc\nds\c.nds
move Game D|Other drive
waitmove 30
expect move failed
expect rom Game D|T\pc\gb\d.gb
expect exists T\pc\gb\d.gb
shot conflict
key back
move Game E|Other drive
wait 1500
shot moving
cancelmove
waitmove 60
expect move cancelled
expect rom Game E|T\pc\ps2\e.iso
expect exists T\pc\ps2\e.iso
expect gone T2\ps2\e.iso
expect gone T2\ps2\e.iso.moving
storage
shot storage2
key back
quit
