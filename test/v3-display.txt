# Display settings: the default for every game and a game's own are saved, and a
# core game's picture takes Integer scale, Fit, Stretch and a custom size, live from its Display card.
# Setup: any small GBA file at <root>\roms\Display Test.gba (a header and an endless loop will do),
# IONEA_TESTROOT=<root>, then --data <tmp> --no-import --dummies 10 --windowed --mute --script test\v3-display.txt
wait 900
type Display Test
key accept
wait 900
key accept
wait 1100
romgame Display Test|T\roms\Display Test.gba|mGBA core
# the default: Settings > Look and layout > Games display
displaycard global
expect screen DisplayCard
displayset Picture size|Integer scale
expect saveddisplay global|Size|integer
shot launcher-display-global
key back
wait 500
# the game's own, saved in the library, then back to the default
displaycard Display Test
displayset Picture size|Stretch
expect saveddisplay Display Test|Size|stretch
displayset Picture size|Default (Integer scale)
expect saveddisplay Display Test|Size|none
key back
wait 500
# the game uses the default (Integer scale), then its own, live
select Display Test
key accept
waitshown 30
wait 2500
expect ingame Display Test
expect corelayout integer
homebutton
wait 800
displaycard Display Test
expect screen DisplayCard
displayset Picture size|Fit
expect corelayout fit
displayset Picture size|Stretch
expect corelayout stretch
shot launcher-display-ingame
displayset Picture size|Custom size
expect corelayout other
key back
wait 600
expect saveddisplay Display Test|Size|custom
closegame
wait 3000
expect ingame none
quit
