# PS4 game: hold opens the quick menu with Game settings (FSR, sharpening, frame pacing...) written to shadPS4's
# own user\custom_configs\CUSA00900.json, and Default removes them again. Uninstall puts the game folder and its
# "-patch" folder in the Recycle Bin (here: IONEA_TEST_RECYCLE) and takes the game off the list.
# Setup: test\make-fake-ps4.ps1 <root>, set IONEA_TESTROOT=<root> and IONEA_TEST_RECYCLE=<root>\bin, then
#   --data <tmp> --no-import --esde <root>\esde --emulators <root>\emus --windowed --mute --script test\n4-ps4-tweaks.txt
wait 900
type Tester
key accept
wait 900
key accept
wait 2500
addchoice
wait 500
key accept
wait 700
key accept
wait 800
key accept
wait 1200
expect games 1
expect name Bloodborne
emucontrols shadPS4|on
expect mouselook Bloodborne|yes
expect gamecursor Bloodborne|hidden
hold 750
expect screen HoldMenu
expect holdrows launch,info,tweaks,move,-transfer,remove,uninstall
expect holdfocus launch
wait 300
shot launcher-ps4-hold-menu
key down 2
expect holdfocus tweaks
key accept
wait 500
expect screen TweaksCard
expect tweakrows GPU.fsr_enabled,GPU.present_mode,GPU.hdr_allowed,GPU.rcas_enabled,GPU.rcas_attenuation,GPU.vblank_frequency
expect cardrows Upscaling | Frame pacing | HDR | More | Done
tweakset GPU.fsr_enabled|true
tweakset GPU.present_mode|Immediate
key down 2
wait 300
shot launcher-ps4-tweaks
# three or more values: Select opens the list of them
cardrow Frame pacing
key accept
expect list Default (Low latency)|Low latency|V-Sync|Uncapped (may tear)
expect listfocus Uncapped (may tear)
key back
expect list closed
expect screen TweaksCard
key back
wait 500
expect tweaks saved
expect json T\emus\shadPS4\user\custom_configs\CUSA00900.json|GPU.fsr_enabled|true
expect json T\emus\shadPS4\user\custom_configs\CUSA00900.json|GPU.present_mode|Immediate
expect json T\emus\shadPS4\user\custom_configs\CUSA00900.json|GPU.rcas_enabled|missing
gametweaks Bloodborne
tweakset GPU.fsr_enabled|default
tweakset GPU.present_mode|default
key back
wait 500
expect tweaks saved
expect gone T\emus\shadPS4\user\custom_configs\CUSA00900.json
# Every game: shadPS4's config.json
emutweaks shadPS4
tweakset GPU.rcas_attenuation|500
key back
wait 500
expect json T\emus\shadPS4\user\config.json|GPU.rcas_attenuation|500
# Uninstall: Cancel has focus first.
hold 750
key down 5
expect holdfocus uninstall
key accept
wait 500
expect screen UninstallCard
shot launcher-ps4-uninstall
key accept
wait 400
expect uninstall cancelled
expect dir T\esde\ROMs\ps4\Bloodborne [CUSA00900]
hold 750
key down 5
expect holdfocus uninstall
key accept
wait 500
key up
key accept
wait 1500
expect uninstall done
expect gone T\esde\ROMs\ps4\Bloodborne [CUSA00900]
expect gone T\esde\ROMs\ps4\Bloodborne [CUSA00900]-patch
expect dir T\bin\Bloodborne [CUSA00900]
expect dir T\bin\Bloodborne [CUSA00900]-patch
expect notinlist Bloodborne
expect games 0
# Settings > Input keys go to shadPS4's own key file (the tweaks above made its user folder)
controls ps4
writeps4 shadPS4
wait 200
expect ps4ini shadPS4|cross = e
expect ps4ini shadPS4|r1 = leftbutton
expect ps4ini shadPS4|l1 = rightbutton
expect ps4ini shadPS4|r2 = lshift
expect ps4ini shadPS4|circle = space
expect ps4ini shadPS4|axis_left_y_minus = w
expect ps4ini shadPS4|mouse_to_joystick = right
expect ps4ini shadPS4|mouse_movement_params = 0.35, 1.3, 0.35
setmousespeed 9
writeps4 shadPS4
wait 200
expect ps4ini shadPS4|mouse_movement_params = 0.35, 4.2, 0.35
expect ps4ini shadPS4|Mouse look sensitivity 9 of 10
expect mousecurve ok
expect ps4ini shadPS4|cross = cross
quit
