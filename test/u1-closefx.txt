# Closing a container: the open panel shrinks and fades back to its tile while the home row
# fades in. B / Esc, a click beside the panel and a pad's B all close it the same way, input right after
# closing goes to the home row, and opening it again during the close stops the close cleanly.
# Run: --data <tmp> --no-import --dummies 12 --windowed --mute --script test\u1-closefx.txt
wait 900
type Close Fx
key accept
wait 900
key accept
wait 1100
order name
folderfill Box|5
key left 15
wait 400
expect name folder:Box
# B (keyboard Esc / pad B are both Act.Back)
key accept
wait 700
expect view folder:Box
tap back
expect view home
expect closefx running
expect closeto folder:Box
expect name folder:Box
wait 110
shot launcher-cf-closing
wait 400
expect closefx none
shot launcher-cf-closed
# input right after closing is not lost
key accept
wait 700
expect view folder:Box
tap back
tap right
expect closefx running
expect sel 1
wait 500
expect closefx none
key left
expect name folder:Box
# a click beside the open panel closes it the same way
key accept
wait 700
mouse down 60 1000
expect view home
expect closeto folder:Box
wait 500
expect closefx none
# open again while it is still closing: the close stops, the folder opens normally
key accept
wait 700
expect view folder:Box
tap back
expect closefx running
tap accept
expect pressing yes
wait 600
expect closefx none
expect view folder:Box
wait 300
shot launcher-cf-reopened
tap back
wait 500
expect view home
expect closefx none
quit
